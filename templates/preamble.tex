% ---------- box primitive: solid left rule + fill, via tikz ----------
\newcommand{\rulebox}[4]{%
  % #1 = fill gray, #2 = rule gray, #3 = label (or empty), #4 = content
  \par\noindent
  \begin{tikzpicture}
    \node[fill=#1, text width=\dimexpr\linewidth-24pt\relax,
          inner sep=8pt, anchor=north west,
          rounded corners=1.5pt] (box) at (0,0) {%
      \begin{minipage}{\dimexpr\linewidth-24pt\relax}
        \ifx&#3&\else{\small\bfseries\color{#2}\MakeUppercase{#3}}\\[3pt]\fi
        \raggedright #4
      \end{minipage}%
    };
    \draw[#2, line width=2.4pt] (box.north west) -- (box.south west);
  \end{tikzpicture}
  \par\vspace{0.5em}
}

% dashed-left-rule variant --- marks a place where the lecture notes were
% incomplete and this content fills the gap
\newcommand{\fillbox}[2]{%
  % #1 = label, #2 = content
  \par\noindent
  \begin{tikzpicture}
    \node[fill=gy1, text width=\dimexpr\linewidth-24pt\relax,
          inner sep=8pt, anchor=north west,
          rounded corners=1.5pt] (box) at (0,0) {%
      \begin{minipage}{\dimexpr\linewidth-24pt\relax}
        {\small\bfseries\color{ink}\MakeUppercase{#1}}\\[3pt]
        \raggedright #2
      \end{minipage}%
    };
    \draw[ink, line width=2.4pt, dash pattern=on 5pt off 3pt] (box.north west) -- (box.south west);
  \end{tikzpicture}
  \par\vspace{0.5em}
}

% Plain explanation box --- the default unit of this document: your notes,
% explained in full, in plain words. Light gray fill, solid black rule.
\newcommand{\notebox}[2]{%
  \rulebox{gy1}{black}{#1}{#2}
}

% Worked example box --- an example problem, in the source's style but
% written fresh, never quoted. Boxed all the way round rather than ruled,
% so it visually reads as "try this yourself" material.
\newcommand{\examplebox}[2]{%
  % #1 = title, #2 = content
  \par\noindent
  \begin{tikzpicture}
    \node[draw=ink, line width=0.9pt, fill=white,
          text width=\dimexpr\linewidth-24pt\relax,
          inner sep=9pt, anchor=north west,
          rounded corners=2pt] (box) at (0,0) {%
      \begin{minipage}{\dimexpr\linewidth-24pt\relax}
        {\small\bfseries\color{ink} EXAMPLE --- #1}\\[4pt]
        \raggedright #2
      \end{minipage}%
    };
  \end{tikzpicture}
  \par\vspace{0.5em}
}

% Key-rule box --- the one thing to remember. Inverted: dark fill, white text.
\newcommand{\keybox}[1]{%
  \par\noindent
  \begin{tikzpicture}
    \node[fill=gy4, text=white, text width=\dimexpr\linewidth-24pt\relax,
          inner sep=9pt, anchor=north west,
          rounded corners=2pt] (box) at (0,0) {%
      \begin{minipage}{\dimexpr\linewidth-24pt\relax}
        \color{white}\raggedright #1
      \end{minipage}%
    };
  \end{tikzpicture}
  \par\vspace{0.5em}
}
